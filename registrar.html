function fetchFreshRoster() { const badge = document.getElementById('rosterBadge'); const tbody = document.getElementById('directoryBody'); badge.innerText = '⏳ Syncing Roster...'; var cbName = 'reg_roster_cb_' + Date.now(); var didRespond = false; // Safety timeout so it never hangs silently var timeoutTimer = setTimeout(function() { if (!didRespond) { badge.innerText = '⚠️ Timeout - Click Refresh'; tbody.innerHTML = 'Server took too long to respond. Click "Refresh List" to retry.'; } }, 12000); window[cbName] = function(rawResponse) { didRespond = true; clearTimeout(timeoutTimer); var el = document.getElementById(cbName); if (el) el.remove(); delete window[cbName]; // Unwrap response whether Code.gs returns an Array directly OR { roster: [...] } / { data: [...] } let data = Array.isArray(rawResponse) ? rawResponse : (rawResponse && Array.isArray(rawResponse.roster)) ? rawResponse.roster : (rawResponse && Array.isArray(rawResponse.data)) ? rawResponse.data : null; if (data) { rosterList = data; rosterMap = {}; sectionToTeacherMap = {}; data.forEach(s => { let id = String(s.id || '').trim(); if (id) rosterMap[id.toLowerCase()] = s; if (s.section && s.teacher) { sectionToTeacherMap[String(s.section).trim()] = String(s.teacher).trim(); } }); populateSectionDropdowns(); populateStudentDatalist(); renderDirectory(); badge.innerText = '🟢 ' + rosterList.length + ' Active Students'; } else { console.error("Unexpected roster format from Code.gs:", rawResponse); badge.innerText = '⚠️ Data Format Mismatch'; tbody.innerHTML = 'Connected, but roster data format was unexpected. Check console (F12).'; } }; var script = document.createElement('script'); script.id = cbName; script.src = APPS_SCRIPT_URL + '?callback=' + cbName + '&_=' + Date.now(); script.onerror = function() { didRespond = true; clearTimeout(timeoutTimer); badge.innerText = '🔴 Connection Error'; tbody.innerHTML = 'Failed to reach Google Apps Script. Check deployment permissions.'; }; document.body.appendChild(script); }
